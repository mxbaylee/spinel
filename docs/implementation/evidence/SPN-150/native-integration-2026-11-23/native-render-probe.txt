import assert from 'node:assert/strict';
import { readFileSync } from 'node:fs';
import { createHash } from 'node:crypto';
import config from '/Users/mxbaylee/projects/spinel-spn150/vite.config.ts';
import { startupHtml } from '/Users/mxbaylee/projects/spinel-spn150/src/build/startupHtml.ts';

assert.equal(typeof config, 'function');

assert.equal((html.match(/<strong class="sp-startup-wordmark">spinel<\/strong>/g) ?? []).length, 1);
