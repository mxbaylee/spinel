#### Decode and verify

Download all three files from the same immutable archive commit into a fresh directory. Standard XZ provides the first decoding step; the following Node snippet verifies the compressed and decoded identities, provenance, component inventory and every exact restored component. `restored/` must not already exist. This is a reconstruction aid; it does not authenticate GitHub, execute product scenarios or regenerate omitted raw data.

```sh
xz --decompress --stdout records.json.xz > records.json
node --input-type=module <<'NODE'
import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { readFileSync, mkdirSync, writeFileSync } from 'node:fs';
import { dirname, join } from 'node:path';
const hash = bytes => createHash('sha256').update(bytes).digest('hex');
const manifest = JSON.parse(readFileSync('manifest.json', 'utf8'));
const archive = readFileSync('records.json.xz');
const decoded = readFileSync('records.json');
assert.equal(archive.length, manifest.archive.bytes);
assert.equal(hash(archive), manifest.archive.sha256);
assert.equal(decoded.length, manifest.decoded.bytes);
assert.equal(hash(decoded), manifest.decoded.sha256);
const envelope = JSON.parse(decoded.toString('utf8'));
assert.equal(envelope.schema, 'spinel-final-acceptance-evidence-v1');
assert.equal(envelope.schema, manifest.decoded.schema);
assert.deepEqual(envelope.provenance, manifest.provenance);
assert.deepEqual(envelope.files.map(({ content, ...metadata }) => metadata), manifest.components);
const seen = new Set();
const restored = envelope.files.map(file => {
  assert.equal(file.encoding, 'utf8');
  assert.equal(typeof file.content, 'string');
  assert.equal(typeof file.path, 'string');
  assert.ok(!file.path.startsWith('/') && !file.path.includes('\\'));
  assert.ok(file.path.split('/').every(part => part && part !== '.' && part !== '..'));
  assert.ok(!seen.has(file.path));
  seen.add(file.path);
  const bytes = Buffer.from(file.content, 'utf8');
  assert.equal(bytes.length, file.bytes);
  assert.equal(hash(bytes), file.sha256);
  return { path: file.path, bytes };
});
mkdirSync('restored'); // Requires a fresh destination.
for (const file of restored) {
  const destination = join('restored', file.path);
  mkdirSync(dirname(destination), { recursive: true });
  writeFileSync(destination, file.bytes, { flag: 'wx' });
}
console.log(`Verified and restored ${restored.length} exact UTF-8 inputs.`);
NODE
```
