import assert from 'node:assert/strict';
import { createRequire } from 'node:module';
import { startDiagnosticTrace } from '/Users/mxbaylee/projects/spinel-spn-071-diagnostic/scripts/ci/spn071-trace.ts';
const require = createRequire('/Users/mxbaylee/projects/spinel/package.json');
const { chromium } = require('@playwright/test');
const browser = await chromium.launch({ headless: true });
try {
  const page = await browser.newPage();
  const cdp = await page.context().newCDPSession(page);
  const finish = await startDiagnosticTrace(cdp, '/tmp/spn071-review/trace-transport-smoke.jsonl');
  await page.evaluate(() => {
    performance.mark('spn071-clock-start');
    Array.from({ length: 10000 }, (_, index) => ({ index })).map(x => x.index * 2);
    performance.mark('spn071-clock-end');
  });
  const result = await finish();
  console.log(JSON.stringify(result));
  assert.equal(result.complete, true);
  await cdp.detach();
} finally {
  await browser.close();
}
