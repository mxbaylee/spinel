import { execFileSync, spawn } from 'node:child_process';
const [cmd, ...args] = process.argv.slice(2);
const start = Date.now();
let peak = 0;
const child = spawn(cmd, args, { stdio: 'inherit' });
const monitor = setInterval(() => {
  const rows = execFileSync('ps', ['-axo', 'pid=,ppid=,rss='], { encoding: 'utf8' }).trim().split('\n').map(line => line.trim().split(/\s+/).map(Number));
  const tree = new Set([process.pid]);
  for (let prior = -1; prior !== tree.size;) {
    prior = tree.size;
    for (const [pid, parent] of rows) if (tree.has(parent)) tree.add(pid);
  }
  const rss = rows.reduce((sum, [pid,,rss]) => sum + (tree.has(pid) ? rss : 0), 0);
  peak = Math.max(peak, rss);
  const pressure = process.platform === 'darwin' ? execFileSync('sysctl', ['-n', 'kern.memorystatus_vm_pressure_level'], {encoding:'utf8'}).trim() : '1';
  if (rss > 2 * 1024 ** 2 || pressure !== '1' || Date.now() - start > 475_000) {
    console.error(`WATCHDOG ABORT: pressure=${pressure}, RSS=${rss/1024}MiB, elapsed=${Date.now()-start}ms`);
    process.kill(0, 'SIGKILL');
  }
}, 250);
child.on('error', error => {clearInterval(monitor); console.error(error); process.exitCode=1;});
child.on('exit', (code, signal) => {clearInterval(monitor);console.error(`WATCHDOG: exit=${code}, signal=${signal}, elapsed=${Date.now()-start}ms, peak=${(peak/1024).toFixed(1)}MiB`);process.exitCode=code??1;});
